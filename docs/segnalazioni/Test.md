`; } catch (e) { bar.innerHTML = `[⚽ Apri il calendario](https://github.com/aandroide/Mandrakodi-Wiki/blob/main/docs/${PAGINA_CALENDARIO})`; console.error("Errore conteggio calendario", e); } } aggiornaBarra(); setInterval(aggiornaBarra, 60000); })(); </script>
